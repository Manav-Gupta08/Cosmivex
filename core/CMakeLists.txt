cmake_minimum_required(VERSION 3.24)
project(universe_core VERSION 0.1.0 LANGUAGES CXX)

option(UOS_BUILD_TESTS "Build native tests" ON)
add_library(universe_core STATIC src/engine.cpp)
target_sources(universe_core PRIVATE telemetry/process.cpp universe/model.cpp events/changes.cpp)
if(WIN32)
  target_sources(universe_core PRIVATE platform/windows/process.cpp)
endif()
target_compile_features(universe_core PUBLIC cxx_std_20)
target_include_directories(universe_core PUBLIC include)
set_target_properties(universe_core PROPERTIES CXX_EXTENSIONS OFF)
if(MSVC)
  target_compile_options(universe_core PRIVATE /W4 /WX /permissive- /EHsc)
else()
  target_compile_options(universe_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
install(TARGETS universe_core ARCHIVE DESTINATION lib)

if(UOS_BUILD_TESTS)
  enable_testing()
  add_executable(universe_core_tests tests/engine_test.cpp)
  target_link_libraries(universe_core_tests PRIVATE universe_core)
  add_test(NAME native_lifecycle_and_mailbox COMMAND universe_core_tests)
  set_tests_properties(native_lifecycle_and_mailbox PROPERTIES TIMEOUT 15)
  add_executable(universe_model_tests tests/model_test.cpp)
  target_link_libraries(universe_model_tests PRIVATE universe_core)
  add_test(NAME universe_hierarchy_and_groups COMMAND universe_model_tests)
  set_tests_properties(universe_hierarchy_and_groups PROPERTIES TIMEOUT 15)
  add_executable(universe_changes_tests tests/changes_test.cpp)
  target_link_libraries(universe_changes_tests PRIVATE universe_core)
  add_test(NAME lifecycle_and_delta_pipeline COMMAND universe_changes_tests)
  set_tests_properties(lifecycle_and_delta_pipeline PROPERTIES TIMEOUT 15)
  if(WIN32)
    add_executable(universe_process_tests tests/process_test.cpp)
    target_link_libraries(universe_process_tests PRIVATE universe_core)
    add_test(NAME windows_process_collection COMMAND universe_process_tests)
    set_tests_properties(windows_process_collection PROPERTIES TIMEOUT 30)
  endif()
endif()
